```python
class AnAcceptanceListIsAnyOrdinaryList(unittest.TestCase):
    """F-0316: `_lift_section` read an acceptance item only as `- [ ] text`, and deleted the rest
    of the block with the block. A note whose `## Acceptance` held four plain bullets reached the
    factory with none of them — not in its acceptance list, not in its Description."""

    def acc(self, section):
        from asf.groom import inbox
        return inbox.parse_inbox_file('# t\nparent: F-0001\n\nBody.\n\n' + section).acceptance

    def test_plain_bullets_are_acceptance_lines(self):
        self.assertEqual(self.acc('## Acceptance\n- the pass binds the free lane\n- a seat is released\n'),
                         ['the pass binds the free lane', 'a seat is released'])

    def test_every_ordinary_list_marker_reads_the_same(self):
        for section in ('## Acceptance\n- a\n- b\n', '## Acceptance\n* a\n* b\n',
                        '## Acceptance\n+ a\n+ b\n', '## Acceptance\n1. a\n2. b\n',
                        '## Acceptance\n1) a\n2) b\n', '## Acceptance\n  - a\n  - b\n'):
            with self.subTest(section=section):
                self.assertEqual(self.acc(section), ['a', 'b'])

    def test_a_checkbox_is_still_read_and_is_not_part_of_the_item(self):
        self.assertEqual(self.acc('## Acceptance\n- [ ] a\n- [x] b\n- [X] c\n'), ['a', 'b', 'c'])

    def test_an_empty_checkbox_is_the_template_placeholder_and_yields_nothing(self):
        """`- [ ]` with no text is what `write_new_item` writes into a fresh card. It says
        nothing, so it is read as nothing — and it is not an unread line either (C4)."""
        from asf.groom import inbox
        c = inbox.parse_inbox_file('# t\n\nBody.\n\n## Acceptance\n- [ ]\n')
        self.assertEqual((c.acceptance, c.description), ([], 'Body.'))

    def test_the_heading_is_read_at_any_depth_and_in_any_case(self):
        for heading in ('## Acceptance', '### Acceptance', '#### acceptance', '## ACCEPTANCE'):
            with self.subTest(heading=heading):
                self.assertEqual(self.acc(heading + '\n- a\n'), ['a'])

    def test_a_nested_heading_ends_the_block(self):
        """Depth-2-only (`## `) swallowed a `### …` and everything under it into the Acceptance
        block, where every line of it was dropped."""
        from asf.groom import inbox
        c = inbox.parse_inbox_file('# t\n\nBody.\n\n## Acceptance\n- a\n\n### Notes\nWhy it matters.\n')
        self.assertEqual(c.acceptance, ['a'])
        self.assertIn('### Notes', c.description)
        self.assertIn('Why it matters.', c.description)

    def test_a_hash_that_is_not_a_heading_does_not_end_the_block(self):
        """ATX needs a space, a tab or the line's end after the hashes, so `#tag` and a
        seven-hash line are not headings. A grammar that ended the block on them left the
        bullets under them out of the acceptance list (F-0316 C3)."""
        for line in ('#1 priority', '#deliverypass', '####### not a heading'):
            with self.subTest(line=line):
                self.assertEqual(self.acc(f'## Acceptance\n- a\n{line}\n- b\n'), ['a', 'b'])

    def test_a_features_list_reads_the_same_markers(self):
        from asf.groom import inbox
        for section in ('## Features\n- Plan tiers\n- Invoicing\n', '## Features\n* Plan tiers\n* Invoicing\n'):
            with self.subTest(section=section):
                c = inbox.parse_inbox_file('# t\n\nBody.\n\n' + section)
                self.assertEqual(c.features, ['Plan tiers', 'Invoicing'])

    def test_an_acceptance_list_types_the_card_a_story_under_its_feature(self):
        """The widened grammar is read by `shape.derive` too: plain bullets under a Feature are
        the acceptance list a Story is, where the card used to be asked for one it already had."""
        from asf.groom import inbox
        c = inbox.parse_inbox_file('# Add plan tiers\nparent: F-0001\n\n## Acceptance\n- tiers render\n')
        result = shape.derive(c, seeded_canonical())
        self.assertNotIsInstance(result, shape.Question, getattr(result, 'text', ''))
        self.assertEqual((result.type, result.rule), ('story', 'parent-feature'))
```

```python
class NothingTheNoteCarriedIsDropped(unittest.TestCase):
    """F-0316 C4: a line intake cannot read is left where the operator wrote it. Before, every
    non-bullet line of a lifted block was deleted with the block — silently, and for good."""

    def read(self, text):
        from asf.groom import inbox
        return inbox.parse_inbox_file(text)

    def test_a_prose_line_in_the_block_stays_in_the_body_under_its_heading(self):
        c = self.read('# t\n\nBody.\n\n## Acceptance\nthe pass binds the free lane\n')
        self.assertEqual(c.acceptance, [])
        self.assertIn('## Acceptance', c.description)
        self.assertIn('the pass binds the free lane', c.description)

    def test_a_block_part_list_part_prose_keeps_both_halves(self):
        c = self.read('# t\n\nBody.\n\n## Acceptance\n- a\nand it must hold on reruns.\n- b\n')
        self.assertEqual(c.acceptance, ['a', 'b'])
        self.assertIn('and it must hold on reruns.', c.description)
        self.assertNotIn('- a', c.description)

    def test_a_block_read_whole_leaves_no_heading_behind(self):
        c = self.read('# t\n\nBody.\n\n## Acceptance\n- a\n- b\n')
        self.assertEqual((c.acceptance, c.description), (['a', 'b'], 'Body.'))

    def test_every_non_blank_line_of_a_note_reaches_the_card_somewhere(self):
        """The invariant, stated as a test: acceptance, a header, or the Description — one of the
        three, for every line the operator wrote."""
        note = ('# delivery pass binds a seat to a busy lane\n'
                'parent: E-0001\n\n'
                'The delivery pass binds a seat to lane-2 while lane-1 is free.\n\n'
                '## Acceptance\n'
                '- the pass binds the free lane\n'
                '- a bound seat is released when the session fails\n\n'
                'Severity: S1\n')
        c = self.read(note)
        seen = set(c.acceptance) | {c.title} | {f'{k}: {v}' for k, v in c.headers.items()}
        seen |= {l.strip() for l in c.description.split('\n')}
        for line in (l.strip() for l in note.split('\n')):
            if not line or line.startswith('## '):
                continue
            with self.subTest(line=line):
                self.assertTrue(
                    any(line == s or line.lstrip('#- ').strip() == s or line.lower() == s.lower()
                        for s in seen),
                    f'{line!r} reached neither the acceptance list, nor a header, nor the body')
```

```python
class TheMintedCardCarriesWhatTheNoteSaid(unittest.TestCase):
    """F-0316, end to end through `asf groom`: the note shape that became F-0315 with an empty
    `## Acceptance` and no severity at all. The four lines and the grade are on the card."""

    def setUp(self):
        self.root = make_record()
        seed(self.root)

    def tearDown(self):
        shutil.rmtree(self.root, ignore_errors=True)

    NOTE = ("# delivery pass binds a seat to a busy lane\n"
            "parent: E-0001\n"
            "\nThe delivery pass binds a seat to lane-2 while lane-1 is free.\n"
            "\n## Acceptance\n"
            "- the pass binds the free lane\n"
            "- a bound seat is released when the session fails\n"
            "- the row names the lane it bound\n"
            "- a second pass over the same seat changes nothing\n"
            "\nSeverity: S1\n")

    def _groom(self, note, name='delivery-pass-binds.md'):
        with open(os.path.join(self.root, 'inbox', name), 'w', encoding='utf-8') as f:
            f.write(note)
        r = run(['groom'], self.root)
        self.assertEqual(r.returncode, 0, r.stderr)
        return r

    def test_the_four_acceptance_lines_are_on_the_minted_card(self):
        self._groom(self.NOTE)
        with open(os.path.join(self.root, 'features', 'F-0002.md'), encoding='utf-8') as f:
            text = f.read()
        for line in ('- [ ] the pass binds the free lane',
                     '- [ ] a bound seat is released when the session fails',
                     '- [ ] the row names the lane it bound',
                     '- [ ] a second pass over the same seat changes nothing'):
            self.assertIn(line + '\n', text)
        self.assertNotIn('- [ ] \n', text)

    def test_the_severity_the_note_declared_is_on_the_card_too(self):
        """A Feature has no `severity:` field (`asf.record.new.SETTABLE`), so the line the
        operator wrote is in the Description — never dropped, and never written as a typed field
        `review_tier` would re-tier the item by (C6)."""
        self._groom(self.NOTE)
        with open(os.path.join(self.root, 'features', 'F-0002.md'), encoding='utf-8') as f:
            meta, body = frontmatter.parse(f.read(), path='features/F-0002.md')
        self.assertIsNone(meta.get('severity'))
        self.assertIn('Severity: S1', body.split('## Acceptance')[0])

    def test_the_card_still_passes_index_and_check(self):
        self._groom(self.NOTE)
        self.assertEqual(run(['index'], self.root).returncode, 0)
        r = run(['check'], self.root)
        self.assertEqual(r.returncode, 0, r.stdout)

    def test_a_minted_bug_carries_the_grade_as_its_field_and_not_twice(self):
        """The same severity line, on a note that mints a Bug: it is the typed field, and the
        Description does not also carry it — a field written twice is the thing `graded` already
        refuses for a title prefix."""
        self._groom("# Checkout is broken\nparent: E-0001\nsignature: test_pay\n"
                    "\nCustomers cannot pay.\n\nSeverity: S1\n", name='checkout.md')
        with open(os.path.join(self.root, 'bugs', 'B-0001.md'), encoding='utf-8') as f:
            meta, body = frontmatter.parse(f.read(), path='bugs/B-0001.md')
        self.assertEqual(meta['severity'], 'S1')
        self.assertNotIn('Severity: S1', body)

    def test_a_note_with_no_severity_mints_exactly_what_it_did_before(self):
        """The control: nothing is added to a card whose note graded nothing."""
        self._groom("# delivery pass binds a seat to a busy lane\nparent: E-0001\n"
                    "\nThe pass binds the wrong lane.\n\n## Acceptance\n- [ ] a\n")
        with open(os.path.join(self.root, 'features', 'F-0002.md'), encoding='utf-8') as f:
            body = f.read()
        self.assertNotIn('Severity', body)
        self.assertIn('- [ ] a\n', body)
```

```python
class ASeverityWrittenBelowTheHeaderBlock(unittest.TestCase):
    """F-0316: the header block stops at the first body line, so a `Severity: S1` the operator
    wrote under the body — or inside the `## Acceptance` block, where it was deleted with the
    block — was never read. `severity:` is the one key read from below the body: its value is the
    closed set `inbox.SEVERITIES`, so prose cannot be mistaken for it (C2)."""

    def read(self, text):
        return inbox.parse_inbox_file(text)

    def test_a_trailing_severity_line_is_the_field_and_leaves_the_body(self):
        c = self.read('# t\n\nThe pass binds the wrong lane.\n\nSeverity: S1\n')
        self.assertEqual(inbox.severity_of(c), 'S1')
        self.assertEqual(c.description, 'The pass binds the wrong lane.')

    def test_a_severity_line_inside_the_acceptance_block_is_read_and_the_list_survives(self):
        """The F-0315 shape: the grade sat under the bullets, inside the lifted block."""
        c = self.read('# t\n\nBody.\n\n## Acceptance\n- a\n- b\n\nSeverity: S1\n')
        self.assertEqual((inbox.severity_of(c), c.acceptance, c.description),
                         ('S1', ['a', 'b'], 'Body.'))

    def test_case_and_spacing_read_the_same(self):
        for line in ('Severity: S1', 'severity: s1', 'SEVERITY:  S1  ', '  severity:\tS1'):
            with self.subTest(line=line):
                self.assertEqual(inbox.severity_of(self.read(f'# t\n\nBody.\n\n{line}\n')), 'S1')

    def test_a_header_block_severity_still_wins_and_the_body_line_stays(self):
        """Two answers and no guess: the declared header is the field, and the line that
        disagrees is left in the body for a person to read (C5)."""
        c = self.read('# t\nseverity: S2\n\nBody.\n\nSeverity: S1\n')
        self.assertEqual(inbox.severity_of(c), 'S2')
        self.assertIn('Severity: S1', c.description)

    def test_two_body_lines_that_disagree_read_as_neither_and_both_stay(self):
        c = self.read('# t\n\nBody.\n\nSeverity: S1\n\nMore.\n\nSeverity: S2\n')
        self.assertEqual(inbox.severity_of(c), 'S3')
        self.assertIn('Severity: S1', c.description)
        self.assertIn('Severity: S2', c.description)

    def test_two_body_lines_that_agree_read_as_one_and_both_go(self):
        c = self.read('# t\n\nBody.\n\nSeverity: S1\n\nMore.\n\nseverity: s1\n')
        self.assertEqual(inbox.severity_of(c), 'S1')
        self.assertEqual(c.description, 'Body.\n\nMore.')

    def test_a_line_outside_the_grammar_is_prose_and_stays_prose(self):
        for line in ('Severity: unclear, maybe S1', 'Severity: S4', 'Severity:',
                     'the Severity: S1 we agreed', 'Severity: S1 if it recurs'):
            with self.subTest(line=line):
                c = self.read(f'# t\n\nBody.\n\n{line}\n')
                self.assertEqual(inbox.severity_of(c), 'S3')
                self.assertIn(line, c.description)

    def test_no_severity_line_changes_nothing(self):
        c = self.read('# t\n\nBody.\n\nMore body.\n')
        self.assertEqual((inbox.severity_of(c), c.headers, c.description),
                         ('S3', {}, 'Body.\n\nMore body.'))

    def test_a_title_prefix_still_grades_a_card_whose_body_says_nothing(self):
        c = inbox.graded(self.read('# S2: the pass binds the wrong lane\n\nBody.\n'))
        self.assertEqual((c.title, inbox.severity_of(c)), ('the pass binds the wrong lane', 'S2'))

    def test_the_body_line_beats_the_title_prefix_as_a_header_line_does(self):
        """One grammar, one precedence: a `severity:` line is a `severity:` line wherever it
        sits, and a header beats the prefix (`test_a_severity_header_wins_...`)."""
        c = inbox.graded(self.read('# S2: the pass binds the wrong lane\n\nBody.\n\nSeverity: S1\n'))
        self.assertEqual((c.title, inbox.severity_of(c)), ('the pass binds the wrong lane', 'S1'))


class ThreeViewsOfOneGrade(unittest.TestCase):
    """F-0316 C7, which is C5 of `asf.groom.inbox`: a card cannot be S1 in one view and S3 in
    another. The mint, the groom line and the stuck-S1 line all read the grade through
    `parse_inbox_file`, so a body-written severity reaches all three or none."""

    def setUp(self):
        self.root = tempfile.mkdtemp(prefix='sev_below_test_')
        os.makedirs(os.path.join(self.root, 'inbox'), exist_ok=True)
        with open(os.path.join(self.root, 'inbox', 'a.md'), 'w', encoding='utf-8') as f:
            f.write('# p1-e2e is red on main\n\nIt has been red since Tuesday.\n\n'
                    'Severity: S1\n\n## Question\n' + EPIC_QUESTION + '\n')

    def tearDown(self):
        shutil.rmtree(self.root, ignore_errors=True)

    def test_the_stuck_card_is_said_aloud_at_s1(self):
        lines = inbox.stuck_s1_lines(self.root, 'inbox')
        self.assertEqual(len(lines), 1, lines)
        self.assertIn('S1 intake card inbox/a.md', lines[0])

    def test_the_groom_line_carries_the_same_token(self):
        lines = inbox.question_lines(self.root, 'inbox')
        self.assertEqual(len(lines), 1, lines)
        self.assertIn('inbox:a.md S1 p1-e2e is red on main', lines[0])

    def test_the_question_block_is_not_read_as_the_body(self):
        """`_split_question` cuts the question before the card is parsed, so the severity is
        read from the card's own body and the question text is not scanned for one."""
        with open(os.path.join(self.root, 'inbox', 'a.md'), encoding='utf-8') as f:
            body, question = inbox._split_question(f.read())
        self.assertEqual(question, EPIC_QUESTION)
        self.assertEqual(inbox.severity_of(inbox.parse_inbox_file(body)), 'S1')
```
