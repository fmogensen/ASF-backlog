   ```python
   def action_cell(row):
       if not row.launches:
           return row.action
       if row.kind == R.BACK_REBASE:
           why = f' ({row.hold_why})' if row.hold_why else ''
           return f"{R.LAUNCH} {row.brief_kind} on {row.branch} — land it as written{why}"
       m = _CHECKS_RED_RE.match(row.correction or '') if row.brief_kind == 'correct' else None
       red = f" — against the red check(s): {m.group(1)}" if m else ''
       return f"{R.LAUNCH} {row.brief_kind} on {row.branch}{red}"
   ```
