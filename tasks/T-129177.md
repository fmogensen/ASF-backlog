A1 — the pure guard: an unread review spends no round (no git). Uses the module's own
`facts()`/`rec()` helpers. The **first** and **fifth** fail and the **sixth** errors on the
unmodified tree — three discriminate. The other three are the controls P11 cause A and cause B
cost: they pass on both sides, and they are the reason the guard is keyed on `is False` and sits
where it sits.

```python
class AnUnreadReviewIsNotAMissingReview(unittest.TestCase):
    """F-0320: ``Lane.branch_facts`` reads a branch's review only from some prior states, and
    ``review_reason`` read the resulting absence as "no ASF review yet" — round 1, a fresh review
    round for code no session touched. A pass that did not read the review wants no round."""

    def test_an_unread_review_keeps_the_branch_where_it_was(self):
        f = facts(review_required=True, review_read=False)      # it looked at nothing
        self.assertEqual(lane.next_state(rec(lane.PR_OPEN, reason='PR #7'), f),
                         (lane.PR_OPEN, 'PR #7'))
        self.assertEqual(lane.next_state(rec(lane.REVIEW, reason='round 3 wanted: x'), f),
                         (lane.REVIEW, 'round 3 wanted: x'))

    def test_a_read_review_that_found_none_still_wants_round_one(self):
        f = facts(review_required=True, review_read=True)
        self.assertEqual(lane.next_state(rec(lane.PR_OPEN), f),
                         (lane.REVIEW, 'round 1 wanted: no ASF review yet'))

    def test_facts_silent_about_the_read_answer_as_they_did_before(self):
        """The tri-state's third leg (P11 cause A): no key at all is not a skipped read. Every
        hand-built facts dict in this module is such a dict."""
        f = facts(review_required=True)                          # no 'review_read' key at all
        self.assertEqual(lane.next_state(rec(lane.PR_OPEN), f),
                         (lane.REVIEW, 'round 1 wanted: no ASF review yet'))
        self.assertEqual(lane.review_reason(f), (1, 'no ASF review yet'))

    def test_a_branch_wanting_no_review_is_gated_whatever_the_read_says(self):
        """P11 cause B: a docs or plan branch reads no review because none is wanted. The guard
        must not reach it — above the `review: none` arm it stalled every one at PR_OPEN."""
        for read in (False, True, None):
            f = facts(review_read=read)                          # review_required False
            self.assertEqual(lane.next_state(rec(lane.PR_OPEN), f),
                             (lane.GATE, 'review: none'), read)

    def test_review_reason_says_which_of_the_three_it_is(self):
        self.assertEqual(lane.review_reason(facts(review_required=True, review_read=False)),
                         (0, 'the review was not read this pass'))
        self.assertEqual(lane.review_reason(facts(review_required=True, review_read=True)),
                         (1, 'no ASF review yet'))
        self.assertEqual(lane.review_reason(facts(review_required=True)),
                         (1, 'no ASF review yet'))

    def test_reads_review_is_every_state_but_landed_or_reaped(self):
        for state in (None, lane.PUSHED, lane.BACK, lane.PR_OPEN, lane.REVIEW, lane.GATE,
                      lane.WAITING_CI, lane.WAITING, lane.QUEUED, lane.MERGING, lane.PARKED,
                      lane.STALE):
            self.assertTrue(lane.reads_review(state, HEAD), state)
        self.assertFalse(lane.reads_review(lane.MERGED, HEAD))
        self.assertFalse(lane.reads_review(lane.REAPED, HEAD))
        self.assertFalse(lane.reads_review(lane.PR_OPEN, None), 'no head, nothing to read it at')
```

A2 — the gate: the review is read whatever the prior state was. Fails today for `GATE`,
`WAITING_CI`, `WAITING`, `QUEUED`, `MERGING` and `STALE`, and `test_a_landed_record_reads_nothing`
fails too: today the key is absent, not `False`. Two tests, both discriminating. The review file is
shared with A3 and A4, which is why its check table and its C list are fields: the verdict comes
from the table (`asf.reviews.verdict`) and a `fail` row with nothing asked reads back as `APPROVED`
(P12).

```python
REVIEW_FILE = """# Review

head: {head}

verdict: approved

| check | result | evidence |
| --- | --- | --- |
| the diff stays inside `writes:` | pass | ok |
| every Step of the Task is implemented | {step_row}
| the acceptance tests are byte-identical to the plan's | pass | ok |
| those tests were run and are green | pass | ok |
| the Gate commands are green | pass | ok |
| no secret value printed, no background process, no skipped check | pass | ok |
{c_list}"""
#: The verdict is derived from the check table first (:mod:`asf.reviews` owns that grammar), so a
#: `changes` review is one with a failing row, not a `verdict:` line that says so. It also needs a
#: C list naming a point: a `fail` row the reviewer asked nothing for reads back as `APPROVED`
#: (:func:`asf.evidence.review.verdict_of`), and the branch would gate on the approval instead.
APPROVES = {'step_row': 'pass | ok |', 'c_list': ''}
ASKS_CHANGES = {'step_row': 'fail | C1 |',
                'c_list': '\n## C\n\n- C1: name the thing this review asks for.\n'}


class TheReviewIsReadFromEveryLivingState(LaneFixture):
    """F-0320: the review a branch carries is read on every pass that could carry the branch back
    to PR_OPEN/REVIEW — the lane's own rebuild moves the head out of WAITING_CI/GATE/QUEUED, and a
    reopened PR brings it back from STALE. It was read only from None/PUSHED/BACK/PR_OPEN/REVIEW,
    so those passes had no verdict to carry."""

    B = 'worker/T-0001'

    def product(self, **conv):
        return super().product(lane={'review': {'code': 'required'}}, **conv)

    def setUp(self):
        super().setUp()
        self.push_lane(self.B, {'a.txt': 'a\n'}, 'feat(T-0001): a')
        code = sh(['git', 'rev-parse', self.B], cwd=self.worker).stdout.strip()
        self.write(self.worker, 'reviews/1-t-0001.md',
                   REVIEW_FILE.format(head=code, **APPROVES))
        sh(['git', 'add', '-A'], cwd=self.worker)
        sh(['git', 'commit', '-qm', 'review(T-0001): round 1'], cwd=self.worker, env_=self.ident)
        sh(['git', 'push', '-q', '-f', 'origin', self.B], cwd=self.worker)
        self.session('coder-t-0001', 'T-0001', self.B)
        sh(['git', 'fetch', '-q', 'origin'], cwd=self.repo)
        self.tip = sh(['git', 'rev-parse', f'origin/{self.B}'], cwd=self.repo).stdout.strip()

    def at(self, state):
        harvest.mark_session(self.state_dir, 'coder-t-0001', branch=self.B, lane={
            'state': state, 'head': self.tip, 'item': 'T-0001', 'pr': None,
            'at': '2026-10-07T08:33:09Z', 'head_at': '2026-10-07T08:05:44Z', 'reason': 'x'})
        ln = lane.Lane(self.product(), self.state_dir, out=lambda *_: None)
        return ln.gather(prs=False)[self.B]

    def test_every_living_prior_state_carries_the_approval(self):
        for state in (lane.PUSHED, lane.BACK, lane.PR_OPEN, lane.REVIEW, lane.GATE,
                      lane.WAITING_CI, lane.WAITING, lane.QUEUED, lane.MERGING, lane.STALE):
            f = self.at(state)
            self.assertTrue(f.get('review_read'), state)
            rv = f.get('review') or {}
            self.assertEqual((rv.get('verdict'), rv.get('current')),
                             (review.APPROVED, True), state)

    def test_a_landed_record_reads_nothing(self):
        f = self.at(lane.MERGED)
        self.assertIs(f.get('review_read'), False)   # set, and False: it did not look
        self.assertFalse(f.get('review'))
```

`assertIs(..., False)` and not `assertFalse`: on the branch this test builds — a finished branch
ahead of the trunk whose *record* reads `MERGED` — the pass reaches edit 2's assignment, so a
missing key is a defect this test must catch rather than pass over. It is not a claim that
`branch_facts` sets the key on every branch in the factory; it does not (**PD5**).

```python
class ARebuiltGateStateBranch(LaneFixture):
    """The fixture of A3 and A4, and the production shape of P8/P10: a branch carrying one commit
    whose patch the trunk also holds, so `drop_copies` rebuilds and pushes it in the same pass,
    and a round-1 review of the pre-rebuild head committed on the branch. ``VERDICT`` is what that
    review says. No test of its own."""

    B = 'worker/T-0001'
    VERDICT = APPROVES

    def product(self, **conv):
        return super().product(lane={'review': {'code': 'required'}}, **conv)

    def setUp(self):
        super().setUp()
        self.push_lane(self.B, {'a.txt': 'a\n'}, 'feat(T-0001): a')
        self.write(self.worker, 'c.txt', 'c\n')
        sh(['git', 'add', '-A'], cwd=self.worker)
        sh(['git', 'commit', '-qm', 'chore: c (#99)'], cwd=self.worker, env_=self.ident)
        sh(['git', 'push', '-q', '-f', 'origin', self.B], cwd=self.worker)
        code = sh(['git', 'rev-parse', self.B], cwd=self.worker).stdout.strip()
        self.write(self.worker, 'reviews/1-t-0001.md',
                   REVIEW_FILE.format(head=code, **self.VERDICT))
        sh(['git', 'add', '-A'], cwd=self.worker)
        sh(['git', 'commit', '-qm', 'review(T-0001): round 1'], cwd=self.worker, env_=self.ident)
        sh(['git', 'push', '-q', '-f', 'origin', self.B], cwd=self.worker)
        self.push_main({'c.txt': 'c\n'}, 'chore: c (#99)')   # the branch now carries a copy
        self.session('coder-t-0001', 'T-0001', self.B)
        sh(['git', 'fetch', '-q', 'origin'], cwd=self.repo)
        self.tip = sh(['git', 'rev-parse', f'origin/{self.B}'], cwd=self.repo).stdout.strip()

    def record(self, state):
        harvest.mark_session(self.state_dir, 'coder-t-0001', branch=self.B, lane={
            'state': state, 'head': self.tip, 'item': 'T-0001', 'pr': None,
            'at': '2026-10-07T08:33:09Z', 'head_at': '2026-10-07T08:05:44Z', 'reason': 'x'})


class ARebuildFromAGateStateKeepsTheApproval(ARebuiltGateStateBranch):
    """F-0320, the reported shape: approved at its head, held in WAITING_CI for heavy CI, and the
    lane's own `drop_copies` rebuilds it on the trunk in a later pass. A product's B-1496 bought a
    fresh review round after each of three such rebuilds and stopped at the relaunch cap; T-0422
    went back to round 1. The rebuilt head carries the same own patch, so the approval stands."""

    VERDICT = APPROVES

    def test_the_rebuild_from_waiting_ci_reaches_the_gate_not_a_review_round(self):
        self.record(lane.WAITING_CI)
        lines = []
        lane.lane_pass(self.product(), self.state_dir, out=lines.append)
        rec_ = self.lane_of(self.B)
        self.assertTrue(any('dropped 1 trunk copies' in l for l in lines), lines)
        self.assertNotEqual(rec_['head'], self.tip, 'the lane rebuilt the branch')
        self.assertEqual(rec_['state'], lane.GATE, lines)
        self.assertIn('approved its head', rec_['reason'])
        self.assertIsNone(rec_.get('round'))
        self.assertFalse(any('no ASF review yet' in l for l in lines), lines)

    def test_a_second_pass_does_not_ask_again(self):
        """A control, measured: it passes today too. Today's first pass writes
        `REVIEW / round 1 wanted: no ASF review yet`, and REVIEW *is* inside the old tuple, so the
        second pass reads the review and recovers to `GATE (reviews/1-t-0001.md approved its
        head)`. That is why B-1496 re-reviewed instead of stalling: each rebuild bought one real
        round — the feeder draws its review row from the record the first pass wrote, before any
        second pass — and the lane then recovered, until the next rebuild reset it again (P8). The
        test is here so the fix is not allowed to introduce a *new* ask on a later pass."""
        self.record(lane.WAITING_CI)
        lane.lane_pass(self.product(), self.state_dir, out=lambda *_: None)
        more = []
        lane.lane_pass(self.product(), self.state_dir, out=more.append)
        self.assertNotEqual(self.lane_of(self.B)['state'], lane.REVIEW, more)
        self.assertFalse(any('wanted' in l for l in more), more)

    def test_the_control_from_pr_open_is_unchanged(self):
        self.record(lane.PR_OPEN)
        lane.lane_pass(self.product(), self.state_dir, out=lambda *_: None)
        self.assertEqual(self.lane_of(self.B)['state'], lane.GATE)
```

class AChangesVerdictReadFromAGateStateGoesBack(ARebuiltGateStateBranch):
    """F-0320's second reachable record shape (P12): the widened read also finds, for the first
    time from a gate state, a review that says `changes`. The branch goes back to its session for
    a correction round — where it spent a review round re-reviewing a head a review had already
    answered. Intended: it is the same answer the lane gives from PR_OPEN today."""

    VERDICT = ASKS_CHANGES

    def test_a_changes_verdict_sends_the_rebuilt_head_back_not_to_a_review_round(self):
        self.record(lane.WAITING_CI)
        lines = []
        lane.lane_pass(self.product(), self.state_dir, out=lines.append)
        rec_ = self.lane_of(self.B)
        self.assertEqual((rec_['state'], rec_['reason']), (lane.BACK, 'kind=review'), lines)
        self.assertIsNone(rec_.get('round'))
        self.assertFalse(any('no ASF review yet' in l for l in lines), lines)

    def test_a_waived_code_branch_stops_landing_over_a_changes_verdict(self):
        """The same widening closes T-0571's hole for the gate states: `review_unwaived`
        (`lane.py:1915-1920`) was unreachable from WAITING_CI, so a waived CODE branch gated as
        `review: none` over a review of its own head that said changes."""
        self.record(lane.WAITING_CI)
        waived = LaneFixture.product(self, lane={'review': {'code': 'none'}})
        lane.lane_pass(waived, self.state_dir, out=lambda *_: None)
        rec_ = self.lane_of(self.B)
        self.assertEqual((rec_['state'], rec_['reason']), (lane.BACK, 'kind=review'))

    def test_the_control_from_pr_open_is_unchanged(self):
        """A control: from PR_OPEN the review was always read, so BACK is what it already did."""
        self.record(lane.PR_OPEN)
        lane.lane_pass(self.product(), self.state_dir, out=lambda *_: None)
        self.assertEqual(self.lane_of(self.B)['state'], lane.BACK)
